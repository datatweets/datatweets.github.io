A/B Testing & ExperimentationModule 4 · Lesson 1
two_prop_ztest.py
def two_prop_ztest(c1, n1, c2, n2):
    p1, p2 = c1 / n1, c2 / n2
    p_pool = (c1 + c2) / (n1 + n2)
    se = np.sqrt(p_pool * (1 - p_pool) * (1 / n1 + 1 / n2))
    return p2 - p1, (p2 - p1) / se

diff, z = two_prop_ztest(503, 5000, 613, 5000)
print(f"difference = {diff:.4f}   z = {z:.3f}")
Outputdifference = 0.0220 z = 3.493

statistically significant, not noise.
